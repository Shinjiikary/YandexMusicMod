// Конвертация заблокированных ссылок raw.githubusercontent.com в рабочие CDN-зеркала
function buildMirrorCandidates(rawUrl) {
  const urls = [];
  if (!rawUrl) return urls;
  const full = rawUrl.includes('://') ? rawUrl : 'https://' + rawUrl;

  const ghMatch = full.match(/^https?:\/\/raw\.githubusercontent\.com\/([^/]+)\/([^/]+)\/([^/]+)\/(.+)$/i);
  if (ghMatch) {
    const [, user, repo, branch, filePath] = ghMatch;
    urls.push(`https://cdn.jsdelivr.net/gh/${user}/${repo}@${branch}/${filePath}`);
    urls.push(`https://raw.githack.com/${user}/${repo}/${branch}/${filePath}`);
  }
  urls.push(full);
  return urls;
}

function downloadUrlBuffer(urlStr) {
  return new Promise((resolve, reject) => {
    const client = urlStr.startsWith('http://') ? http : https;
    const req = client.get(
      urlStr,
      {
        timeout: 15000,
        headers: { 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)' }
      },
      (res) => {
        if ([301, 302, 303, 307, 308].includes(res.statusCode) && res.headers.location) {
          const nextUrl = new URL(res.headers.location, urlStr).href;
          res.resume();
          return downloadUrlBuffer(nextUrl).then(resolve).catch(reject);
        }
        if (res.statusCode !== 200) {
          res.resume();
          return reject(new Error(`HTTP ${res.statusCode}`));
        }
        const chunks = [];
        res.on('data', (c) => chunks.push(c));
        res.on('end', () => resolve(Buffer.concat(chunks)));
      }
    );
    req.on('error', reject);
    req.on('timeout', () => {
      req.destroy();
      reject(new Error('Timeout'));
    });
  });
}

async function downloadWithMirrors(rawUrl) {
  const candidates = Array.isArray(rawUrl) ? rawUrl : buildMirrorCandidates(rawUrl);
  let lastErr = null;
  for (const candidate of candidates) {
    try {
      const buf = await downloadUrlBuffer(candidate);
      if (buf && buf.length > 64) return buf;
    } catch (e) {
      lastErr = e;
    }
  }
  throw lastErr || new Error('All mirrors failed');
}

// Локальный микро-сервер для раздачи чистых MP3 с поддержкой перемотки (HTTP 206 Partial Content)
let localAudioServerPort = 0;
const remoteTrackUrlMap = new Map();

function startLocalAudioServer() {
  return new Promise((resolve) => {
    const server = http.createServer(async (req, res) => {
      res.setHeader('Access-Control-Allow-Origin', '*');
      res.setHeader('Access-Control-Allow-Methods', 'GET, HEAD, OPTIONS');
      res.setHeader('Access-Control-Allow-Headers', '*');

      if (req.method === 'OPTIONS') {
        res.writeHead(204);
        return res.end();
      }

      try {
        const parsed = new URL(req.url, 'http://127.0.0.1');
        const match = parsed.pathname.match(/^\/track\/([0-9A-Za-z_-]+)\.mp3$/);
        if (!match) {
          res.writeHead(404);
          return res.end('Not found');
        }

        const trackId = match[1];
        const cachedFile = path.join(audioCacheDir, `${trackId}.mp3`);
        let audioBuffer = null;

        // 1. Если MP3 уже есть на диске — берём мгновенно с диска
        if (fs.existsSync(cachedFile) && fs.statSync(cachedFile).size > 1024) {
          audioBuffer = fs.readFileSync(cachedFile);
        } else {
          // 2. Иначе скачиваем через зеркала и сохраняем в кэш на ПК
          const remoteUrl = parsed.searchParams.get('src') || remoteTrackUrlMap.get(trackId);
          if (!remoteUrl) {
            res.writeHead(404);
            return res.end('No remote source');
          }
          audioBuffer = await downloadWithMirrors(remoteUrl);
          if (audioBuffer && audioBuffer.length > 1024) {
            try {
              fs.writeFileSync(cachedFile, audioBuffer);
            } catch (e) {}
          }
        }

        if (!audioBuffer || audioBuffer.length === 0) {
          res.writeHead(500);
          return res.end('Empty audio');
        }

        const total = audioBuffer.length;
        const range = req.headers.range;

        if (range) {
          const parts = range.replace(/bytes=/, '').split('-');
          const start = parseInt(parts[0], 10) || 0;
          const end = parts[1] ? parseInt(parts[1], 10) : total - 1;
          const chunk = audioBuffer.subarray(start, end + 1);

          res.writeHead(206, {
            'Content-Range': `bytes ${start}-${end}/${total}`,
            'Accept-Ranges': 'bytes',
            'Content-Length': chunk.length,
            'Content-Type': 'audio/mpeg'
          });
          return res.end(chunk);
        } else {
          res.writeHead(200, {
            'Accept-Ranges': 'bytes',
            'Content-Length': total,
            'Content-Type': 'audio/mpeg'
          });
          return res.end(audioBuffer);
        }
      } catch (err) {
        console.error('Audio server error:', err.message);
        res.writeHead(502);
        res.end('Audio fetch error');
      }
    });

    server.listen(0, '127.0.0.1', () => {
      localAudioServerPort = server.address().port;
      resolve(localAudioServerPort);
    });
  });
}

function findWorkshopDirs() {
  const candidates = new Set([
    'C:\\Program Files (x86)\\Steam',
    'C:\\Program Files\\Steam',
    'C:\\Steam',
    'D:\\Steam',
    'D:\\SteamLibrary',
    'E:\\Steam',
    'E:\\SteamLibrary',
    'F:\\SteamLibrary'
  ]);

  for (const base of Array.from(candidates)) {
    const vdfPath = path.join(base, 'steamapps', 'libraryfolders.vdf');
    if (fs.existsSync(vdfPath)) {
      try {
        const content = fs.readFileSync(vdfPath, 'utf-8');
        const matches = content.matchAll(/"path"\s+"([^"]+)"/g);
        for (const m of matches) {
          candidates.add(m[1].replace(/\\\\/g, '\\'));
        }
      } catch (e) {}
    }
  }

  const workshopDirs = [];
  for (const lib of candidates) {
    const weDir = path.join(lib, 'steamapps', 'workshop', 'content', '431960');
    if (fs.existsSync(weDir)) workshopDirs.push(weDir);
  }
  return workshopDirs;
}

function scanAllWallpapers() {
  const items = [];
  let weCount = 0;
  let customCount = 0;

  const weDirs = findWorkshopDirs();
  for (const weDir of weDirs) {
    try {
      const folders = fs.readdirSync(weDir, { withFileTypes: true });
      for (const dir of folders) {
        if (!dir.isDirectory()) continue;
        const itemPath = path.join(weDir, dir.name);
        const projFile = path.join(itemPath, 'project.json');

        let title = `Wallpaper ${dir.name}`;
        let videoFile = null;
        let previewFile = null;

        if (fs.existsSync(projFile)) {
          try {
            const proj = JSON.parse(fs.readFileSync(projFile, 'utf-8'));
            if (proj.title) title = proj.title;
            if (proj.file) {
              const candidate = path.join(itemPath, proj.file);
              if (fs.existsSync(candidate) && /\.(mp4|webm)$/i.test(proj.file)) {
                videoFile = candidate;
              }
            }
            if (proj.preview) {
              const pCandidate = path.join(itemPath, proj.preview);
              if (fs.existsSync(pCandidate)) previewFile = pCandidate;
            }
          } catch (e) {}
        }

        if (!videoFile) {
          const files = fs.readdirSync(itemPath);
          const foundVid = files.find((f) => /\.(mp4|webm)$/i.test(f));
          if (foundVid) videoFile = path.join(itemPath, foundVid);
          if (!previewFile) {
            const foundPrev = files.find((f) => /^preview\.(jpg|jpeg|gif|png|webp)$/i.test(f));
            if (foundPrev) previewFile = path.join(itemPath, foundPrev);
          }
        }

        if (videoFile) {
          weCount++;
          items.push({
            id: `we_${dir.name}`,
            title,
            source: 'we',
            videoUrl: pathToFileURL(videoFile).href,
            previewUrl: previewFile ? pathToFileURL(previewFile).href : null
          });
        }
      }
    } catch (e) {}
  }

  if (fs.existsSync(customWallpapersDir)) {
    try {
      const files = fs.readdirSync(customWallpapersDir);
      for (const f of files) {
        if (/\.(mp4|webm)$/i.test(f)) {
          customCount++;
          const fullPath = path.join(customWallpapersDir, f);
          items.push({
            id: `custom_${f}`,
            title: path.parse(f).name,
            source: 'custom',
            videoUrl: pathToFileURL(fullPath).href,
            previewUrl: null
          });
        }
      }
    } catch (e) {}
  }

  return { weCount, customCount, items };
}

  function updateWaveCenterOffset() {
    const vibeRoot = document.querySelector('[class*="VibePage_root"]');
    const anchor =
      document.querySelector('[class*="VibePage_playerBlock"]') ||
      document.querySelector('[class*="VibePage_vibeAnimation"]');

    if (!vibeRoot || !anchor) {
      if (currentShiftPx !== 0) {
        currentShiftPx = 0;
        document.documentElement.style.setProperty('--ym-wave-shift', '0px');
      }
      return;
    }

    if (!state.hideVibeCarousel && !state.zenMode) {
      if (currentShiftPx !== 0) {
        currentShiftPx = 0;
        document.documentElement.style.setProperty('--ym-wave-shift', '0px');
      }
      return;
    }

    const rect = anchor.getBoundingClientRect();
    if (rect.width === 0) return;

    const naturalCenterX = (rect.left - currentShiftPx) + rect.width / 2;

    let targetCenterX = window.innerWidth / 2;
    if (!state.zenMode) {
      const navEl = document.querySelector('[class*="NavbarDesktop_root"]');
      const navWidth = navEl ? navEl.getBoundingClientRect().width : 210;
      targetCenterX = Math.round(window.innerWidth / 2 + navWidth * 0.25);
    }

    const desiredShift = Math.round(targetCenterX - naturalCenterX);
    if (Math.abs(desiredShift - currentShiftPx) > 2) {
      currentShiftPx = desiredShift;
      document.documentElement.style.setProperty('--ym-wave-shift', `${currentShiftPx}px`);
    }
  }

  function tagPageElements() {
    const navRoot = document.querySelector('[class*="NavbarDesktop_root"]');
    if (navRoot) {
      const links = navRoot.querySelectorAll('a, button');
      links.forEach(el => {
        const txt = (el.textContent || '').trim();
        const row = el.closest('li') || el;
        if (txt === 'Поиск') row.setAttribute('data-ym-nav', 'search');
        else if (txt.includes('Для вас') || txt.includes('Тренды')) row.setAttribute('data-ym-nav', 'trends');
        else if (txt === 'Концерты') row.setAttribute('data-ym-nav', 'concerts');
        else if (txt.includes('Книги') || txt.includes('подкасты')) row.setAttribute('data-ym-nav', 'podcasts');
        else if (txt === 'Коллекция') row.setAttribute('data-ym-nav', 'collection');
        else if (txt.includes('Ваш Плюс')) row.setAttribute('data-ym-nav', 'plus');
      });
      // 4. Авто-вырезание рекламных поп-апов («Так звучит хоккей / Реклама») и нижних промо-плашек («В другой раз»)
    document.querySelectorAll('[role="dialog"], [class*="Modal_"], [class*="Popup_"], [class*="Notification"], [class*="Banner"], [class*="BarBelow"]').forEach(el => {
      if (el.id === 'ym-wp-modal' || el.id === 'ym-wp-modal-backdrop') return;
      const txt = el.textContent || '';
      if (
        txt.includes('Реклама') ||
        txt.includes('В другой раз') ||
        txt.includes('Расскажите больше') ||
        txt.includes('Слушать плейлист')
      ) {
        const dismissBtn = Array.from(el.querySelectorAll('button')).find(b => {
          const bTxt = (b.textContent || '').trim();
          const aria = b.getAttribute('aria-label') || '';
          return bTxt === 'Закрыть' || bTxt === 'В другой раз' || aria === 'Закрыть';
        });
        if (dismissBtn) dismissBtn.click();
        el.style.setProperty('display', 'none', 'important');
      }
    });
    }

    const progressRow = document.querySelector('[class*="VibePlayerBar_progress"]');
    if (progressRow) {
      Array.from(progressRow.children).forEach(child => {
        const hasTrackInfo = child.querySelector(
          '[class*="VibePlayerbarTimecode"], [class*="VibePlayerbarTrackName"], [class*="VibePlayerbarMeta"]'

function getHeaders(token) {

// Забираем очередь треков "Моей волны"
async function getMyWave(token) {
  const res = await fetch(`${API_BASE}/rotor/station/user:onyourwave/tracks?settings2=true`, {
    headers: getHeaders(token)
  });
  const data = await res.json();
  if (!data.result || !data.result.sequence) throw new Error('Ошибка получения волны');

  return data.result.sequence.map(item => {
    const t = item.track;
    return {
      id: t.id,
      title: t.title,
      artist: t.artists ? t.artists.map(a => a.name).join(', ') : 'Неизвестен',
      cover: t.coverUri ? `https://${t.coverUri.replace('%%', '400x400')}` : ''
    };
  });
}

// Вытаскиваем прямую MP3-ссылку на аудиопоток
async function getTrackUrl(token, trackId) {
  const infoRes = await fetch(`${API_BASE}/tracks/${trackId}/download-info`, {
    headers: getHeaders(token)
  });
  const infoData = await infoRes.json();

  const best = infoData.result
    .filter(x => x.codec === 'mp3')
    .sort((a, b) => b.bitrateInKbps - a.bitrateInKbps)[0] || infoData.result[0];

  const dlRes = await fetch(`${best.downloadInfoUrl}&format=json`, {
    headers: getHeaders(token)
  });
  const dl = await dlRes.json();

  const sign = crypto
    .createHash('md5')
    .update(SIGN_SALT + dl.path.slice(1) + dl.s)
    .digest('hex');

  return `https://${dl.host}/get-mp3/${sign}/${dl.ts}${dl.path}`;
}

class AudioCore {
  constructor() {
    this.audio = new Audio();
    this.audio.crossOrigin = 'anonymous';
    this.audio.volume = 0.5;
  }

  playUrl(url) {
    this.audio.src = url;
    return this.audio.play();
  }

  toggle() {
    if (!this.audio.src) return false;
    if (this.audio.paused) {
      this.audio.play();
      return true;
    } else {
      this.audio.pause();
      return false;
    }
  }

  setVolume(val) {
    this.audio.volume = Math.max(0, Math.min(1, val));
  }

  seek(percent) {
    if (this.audio.duration) {
      this.audio.currentTime = (percent / 100) * this.audio.duration;
    }
  }

  onTimeUpdate(callback) {
    this.audio.addEventListener('timeupdate', () => {
      const progress = this.audio.duration ? (this.audio.currentTime / this.audio.duration) * 100 : 0;
      callback(progress, this.audio.currentTime, this.audio.duration || 0);
    });
  }

  onEnded(callback) {
    this.audio.addEventListener('ended', callback);
  }
}

function applyWallpaperFile(file) {
  if (!file) return;
  const url = URL.createObjectURL(file);

  if (file.type.startsWith('video/')) {
    bgImage.classList.add('hidden');
    bgVideo.classList.remove('hidden');
    bgVideo.src = url;
    bgVideo.play();
  } else if (file.type.startsWith('image/')) {
    bgVideo.classList.add('hidden');
    bgImage.classList.remove('hidden');
    bgImage.style.backgroundImage = `url("${url}")`;
  }

  // Если прозрачность была выкручена в 100%, автоматически делаем панели стеклянными, чтобы обои было видно сразу
  if (Number(opacitySlider.value) > 75) {
    opacitySlider.value = 55;
    document.documentElement.style.setProperty('--panel-opacity', '0.55');
    window.nebulaAPI.setConfig({ hudOpacity: 55 });
  }
}

class VisualEngine {
  constructor() {
    this.canvas = document.getElementById('bgCanvas');
    this.ctx = this.canvas.getContext('2d');
    this.video = document.getElementById('bgVideo');
    this.particles = [];
    this.mouse = { x: -1000, y: -1000 };
    this.analyser = null;
    this.dataArray = null;
    this.audioReactive = true;
    this.visScale = 1.0;

    this.resize();
    window.addEventListener('resize', () => this.resize());
    window.addEventListener('mousemove', (e) => {
      this.mouse.x = e.clientX;
      this.mouse.y = e.clientY;
    });

    this.initParticles(75);
    this.animate();
  }

  resize() {
    this.canvas.width = window.innerWidth;
    this.canvas.height = window.innerHeight;
  }

  initParticles(count) {
    this.particles = [];
    for (let i = 0; i < count; i++) {
      this.particles.push({
        x: Math.random() * this.canvas.width,
        y: Math.random() * this.canvas.height,
        vx: (Math.random() - 0.5) * 0.4,
        vy: (Math.random() - 0.5) * 0.4,
        r: Math.random() * 2 + 1
      });
    }
  }

  connectAudio(audioElement) {
    if (this.analyser) return;
    try {
      const ctx = new (window.AudioContext || window.webkitAudioContext)();
      const src = ctx.createMediaElementSource(audioElement);
      this.analyser = ctx.createAnalyser();
      this.analyser.fftSize = 128;
      src.connect(this.analyser);
      this.analyser.connect(ctx.destination);
      this.dataArray = new Uint8Array(this.analyser.frequencyBinCount);
    } catch (e) {
      console.warn('Web Audio API init skipped:', e);
    }
  }

  setVideoWallpaper(file) {
    const url = URL.createObjectURL(file);
    this.video.src = url;
    this.video.classList.remove('hidden');
    this.video.play();
  }

  getBassEnergy() {
    if (!this.analyser || !this.audioReactive) return 0;
    this.analyser.getByteFrequencyData(this.dataArray);
    let sum = 0;
    for (let i = 0; i < 8; i++) sum += this.dataArray[i];
    return (sum / (8 * 255)); // от 0.0 до 1.0
  }

  animate() {
    requestAnimationFrame(() => this.animate());
    const { ctx, canvas } = this;
    ctx.clearRect(0, 0, canvas.width, canvas.height);

    const bass = this.getBassEnergy();

    // Передаем пульсацию баса на обложку
    const cover = document.getElementById('coverImg');
    if (cover) {
      const scale = 1 + bass * 0.08 * this.visScale;
      cover.style.transform = `scale(${scale.toFixed(3)})`;
    }

    // Рисуем созвездия
    for (let i = 0; i < this.particles.length; i++) {
      const p = this.particles[i];
      p.x += p.vx * (1 + bass * 2);
      p.y += p.vy * (1 + bass * 2);

      if (p.x < 0 || p.x > canvas.width) p.vx *= -1;
      if (p.y < 0 || p.y > canvas.height) p.vy *= -1;

      ctx.beginPath();
      ctx.arc(p.x, p.y, p.r + bass * 2, 0, Math.PI * 2);
      ctx.fillStyle = `rgba(180, 130, 255, ${0.4 + bass * 0.5})`;
      ctx.fill();

      for (let j = i + 1; j < this.particles.length; j++) {
        const p2 = this.particles[j];
        const dist = Math.hypot(p.x - p2.x, p.y - p2.y);
        if (dist < 130) {
          ctx.beginPath();
          ctx.moveTo(p.x, p.y);
          ctx.lineTo(p2.x, p2.y);
          ctx.strokeStyle = `rgba(138, 43, 226, ${(1 - dist / 130) * 0.25})`;
          ctx.stroke();
        }
      }

      const mDist = Math.hypot(p.x - this.mouse.x, p.y - this.mouse.y);
      if (mDist < 160) {
        ctx.beginPath();
        ctx.moveTo(p.x, p.y);
        ctx.lineTo(this.mouse.x, this.mouse.y);
        ctx.strokeStyle = `rgba(0, 210, 255, ${(1 - mDist / 160) * 0.4})`;
        ctx.stroke();
      }
    }

    // Нижний спектральный эквалайзер
    if (this.analyser) {
      const bars = 40;
      const barWidth = (canvas.width * 0.5 * this.visScale) / bars;
      const startX = (canvas.width - bars * barWidth) / 2;

      for (let i = 0; i < bars; i++) {
        const val = this.audioReactive
          ? this.dataArray[i] / 255
          : (Math.sin(Date.now() * 0.003 + i * 0.3) + 1) * 0.15;

        const barHeight = val * 90 * this.visScale;
        const x = startX + i * barWidth;
        const y = canvas.height - 115 - barHeight;

        const grad = ctx.createLinearGradient(x, y, x, canvas.height - 115);
        grad.addColorStop(0, 'rgba(0, 225, 255, 0.75)');
        grad.addColorStop(1, 'rgba(138, 43, 226, 0.05)');

        ctx.fillStyle = grad;
        ctx.beginPath();
        ctx.roundRect(x + 2, y, Math.max(2, barWidth - 4), barHeight, 4);
        ctx.fill();
      }
    }
  }
}
